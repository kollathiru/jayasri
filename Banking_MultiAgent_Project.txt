
import asyncio

from autogen_agentchat.agents import AssistantAgent
from autogen_agentchat.teams import RoundRobinGroupChat
from autogen_agentchat.conditions import TextMentionTermination
from autogen_ext.models.ollama import OllamaChatCompletionClient


# --------------------------------
# 1. Connect Ollama LLM
# --------------------------------

model_client = OllamaChatCompletionClient(
    model="llama3",
    host="http://localhost:11434"
)


# --------------------------------
# 2. Customer Agent
# --------------------------------

customer_agent = AssistantAgent(
    name="Customer_Agent",
    model_client=model_client,
    system_message="""
    You are a Banking Customer Assistant.

    Responsibilities:
    1. Understand customer queries.
    2. Identify customer requirements.
    3. Organize customer and account information.
    4. Share the requirements with other agents.

    Do not request passwords, PINs or OTPs.
    """
)


# --------------------------------
# 3. Transaction Analysis Agent
# --------------------------------

transaction_agent = AssistantAgent(
    name="Transaction_Agent",
    model_client=model_client,
    system_message="""
    You are a Banking Transaction Analysis Agent.

    Responsibilities:
    1. Review banking transaction records.
    2. Analyze debit and credit transactions.
    3. Calculate transaction totals.
    4. Identify unusual transaction patterns.
    5. Explain transaction information.

    Never modify banking records or execute payments.
    """
)


# --------------------------------
# 4. Fraud Detection Agent
# --------------------------------

fraud_agent = AssistantAgent(
    name="Fraud_Detection_Agent",
    model_client=model_client,
    system_message="""
    You are a Banking Fraud Detection Assistant.

    Responsibilities:
    1. Review suspicious transactions.
    2. Identify unusual transaction amounts.
    3. Check unusual locations and transaction frequency.
    4. Highlight potentially suspicious activities.
    5. Recommend contacting the bank when appropriate.

    Do not claim fraud is confirmed.
    Flag suspicious transactions for human review.
    """
)


# --------------------------------
# 5. Banking Support Agent
# --------------------------------

support_agent = AssistantAgent(
    name="Banking_Support_Agent",
    model_client=model_client,
    system_message="""
    You are a Banking Support Assistant.

    Responsibilities:
    1. Review all previous agent findings.
    2. Prepare a clear customer-friendly response.
    3. Explain transaction details.
    4. Provide safe banking recommendations.
    5. Recommend contacting official bank support
       for disputed or suspicious transactions.

    Never request PIN, password or OTP.
    Never execute banking transactions.

    Finish your response with:
    BANKING_RESPONSE_COMPLETE
    """
)


# --------------------------------
# 6. Customer Banking Data
# --------------------------------

banking_data = """
Customer ID: C1001
Customer Name: Ravi Kumar
Account Type: Savings Account
Available Balance: Rs. 45,000

Transaction History:

Transaction 1:
Date: 2026-11-01
Amount: Rs. 2,000
Type: Debit
Location: Hyderabad

Transaction 2:
Date: 2026-11-02
Amount: Rs. 5,000
Type: Debit
Location: Mumbai

Transaction 3:
Date: 2026-11-03
Amount: Rs. 1,500
Type: Credit
Location: Hyderabad

Transaction 4:
Date: 2026-11-04
Amount: Rs. 15,000
Type: Debit
Location: Unknown Location
"""


# --------------------------------
# 7. Multi-Agent Orchestration
# --------------------------------

termination = TextMentionTermination(
    "BANKING_RESPONSE_COMPLETE"
)

team = RoundRobinGroupChat(
    participants=[
        customer_agent,
        transaction_agent,
        fraud_agent,
        support_agent
    ],
    termination_condition=termination
)


# --------------------------------
# 8. Run Banking Application
# --------------------------------

async def main():

    print("\n====================================")
    print("   AI MULTI-AGENT BANKING SYSTEM")
    print("====================================")

    customer_query = input(
        "\nEnter your banking query: "
    )

    task = f"""
    CUSTOMER BANKING QUERY:
    {customer_query}

    CUSTOMER BANKING INFORMATION:
    {banking_data}

    Process this request using all four agents.

    Analyze transactions and identify potential
    suspicious activity.

    Generate a final customer-friendly response.
    """

    result = await team.run(task=task)

    print("\n========== AGENT CONVERSATION ==========")

    for message in result.messages:
        if hasattr(message, "content"):
            print(f"\n{message.source}:")
            print(message.content)

    print("\n========== BANKING SYSTEM COMPLETED ==========")

    await model_client.close()


if __name__ == "__main__":
    asyncio.run(main())
